# Every OpenID Foundation document vendored in references/oidf-sources/ — one
# <local-file>=<url> entry per line, consumed by tools/fetch-specs.sh:
#
#   ./tools/fetch-specs.sh --dir references/oidf-sources --file references/oidf-sources/manifest.txt
#   ./tools/fetch-specs.sh --dir references/oidf-sources --file references/oidf-sources/manifest.txt --check

# DIFFERENT means "upstream changed": review it, then refresh.
